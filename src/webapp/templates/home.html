Get started locally

Install Django Probe and preview the exact payload without sending it:

uv add --dev django-probe
uv run django-probe scan .

Then {% if user.is_authenticated %}create a project{% else %}create an account and project{% endif %}, save its token as the DJANGO_PROBE_TOKEN repository secret, and commit the workflow.

Add it to GitHub Actions

{% verbatim %} 
name: Django Probe
on:
  schedule:
    # Choose a different minute and hour to help spread load on our servers.
    - cron: "17 4 1 * *"
  workflow_dispatch:
jobs:
  submit:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: astral-sh/setup-uv@v10
      - run: uv run django-probe submit .
        env:
          DJANGO_PROBE_TOKEN: ${{ secrets.DJANGO_PROBE_TOKEN }}
